local indexWord = IB_NO_VIRTUALIZE(function(w)
	if wordSet[w] then
		return
	end

	local b = buckets[w:sub(1, 1)]

	if not b then
		b = {}
		buckets[w:sub(1, 1)] = b
	end

	local pos = #b + 1

	for i, v in b do
		if #v > #w then
			pos = i
			break
		end
	end

	table.insert(b, pos, w)
	wordSet[w] = true
	wordCount += 1
end)

local coolrequest = IB_NO_VIRTUALIZE(function(url)
	local zov = {
		{
			name = "request",
			func = function(u)
				local req = request

				if req then
					local res = req({Url = u, Method = "GET"})

					if res.StatusCode ~= 200 then
						error(`status {res.StatusCode}`)
					end

					return res.Body
				end
			end,
		},

		{
			name = "game:HttpGet",
			func = function(u) return game:HttpGet(u) end,
		},
	}

	for _, v in zov do
		local s, src = pcall(v.func, url)

		if s and type(src) == "string" then
			return src
		else
			warn(`{v.name} fail: {src}`)
		end
	end

	return nil
end)

		return tostring(data.version)
	end

	return nil
end

local function loadModule(src)
	if type(src) ~= "string" then
		return nil
	end

	local fn = loadstring(src)

	if not fn then
		return nil
	end

	local s, mod = pcall(fn)

	return s and mod or nil
end

local findWords = IB_NO_VIRTUALIZE(function(prefix, skipTried, includeUsed, ignoreSuffix)
	local b = buckets[prefix:sub(1, 1)]

	if not b then
		return {}
	end

	local sfxs = suffixList()

	if ignoreSuffix or #sfxs == 0 then
		sfxs = {""}
	end

	local groups = {}

	for i = 1, #sfxs do
		groups[i] = {}
	end

	for _, v in b do
		if #v >= #prefix and v:sub(1, #prefix) == prefix and not blacklist[v] and (includeUsed or not usedWords[v]) and (not skipTried or not triedThisRound[v]) then
			for i, sfx in sfxs do
				if #sfx == 0 or (#v >= #sfx and v:sub(-#sfx) == sfx) then
					local g = groups[i]
					g[#g + 1] = v
					break
				end
			end
		end
	end

	for i = 1, #sfxs do
		if #groups[i] > 0 then
			return groups[i]
		end
	end

	return {}
end)

local sortCmp = IB_NO_VIRTUALIZE(function()
	local mode = Options.SuggestSort.Value

	if mode == "Longest" then
		return function(a, b)
			if #a ~= #b then
				return #a > #b
			end

			return a < b
		end
	elseif mode == "a-Z" then
		return function(a, b)
			return a < b
		end
	elseif mode == "Z-a" then
		return function(a, b)
			return a > b
		end
	elseif mode == "Most common" then
		return function(a, b)
			local ra = freq[a] or math.huge
			local rb = freq[b] or math.huge

			if ra ~= rb then
				return ra < rb
			end

			return a < b
		end
	end

	return nil
end)

local orderedCandidates = IB_NO_VIRTUALIZE(function(prefix, includeUsed, skipTried, ignoreSuffix)
	local list = findWords(prefix, skipTried, includeUsed, ignoreSuffix)

	local cmp = sortCmp()
	local out = {}

	if Toggles.ShowWhitelisted.Value then
		local wl = {}
		local rest = {}

		for _, v in list do
			if whitelist[v] then
				wl[#wl + 1] = v
			else
				rest[#rest + 1] = v
			end
		end

		if cmp then
			table.sort(wl, cmp)
			table.sort(rest, cmp)
		end

		for _, v in wl do
			out[#out + 1] = {word = v, check = true}
		end

		for _, v in rest do
			out[#out + 1] = {word = v}
		end
	else
		if cmp then
			table.sort(list, cmp)
		end

		for _, v in list do
			out[#out + 1] = {word = v}
		end
	end

	return out
end)

local adjacentKey = IB_NO_VIRTUALIZE(function(c)
	local target = keys[c]

	if not target then
		return nil
	end

	local near = {}

	for k, v in keys do
		if k ~= c and math.sqrt((v.x - target.x) ^ 2 + (v.y - target.y) ^ 2) <= 1.2 then
			near[#near + 1] = k
		end
	end

	if #near == 0 then
		return nil
	end

	return near[math.random(1, #near)]
end)

local function perCharDelay()
	if Toggles.FixedDelay.Value then
		return Options.TypeDelay.Value
	end

	local lo = math.min(Options.WpmMin.Value, Options.WpmMax.Value)
	local hi = math.max(Options.WpmMin.Value, Options.WpmMax.Value)

	if not sessionWpm then
		sessionWpm = math.random(lo, hi)
		currentWpm = sessionWpm
	end

	local wpm = currentWpm * math.clamp(0.7 + successCount * 0.1, 0.7, 1)

	return 60 / (wpm * 5)
end

local function driftWpm()
	if Toggles.FixedDelay.Value then
		return
	end

	local lo = math.min(Options.WpmMin.Value, Options.WpmMax.Value)
	local hi = math.max(Options.WpmMin.Value, Options.WpmMax.Value)

	if not sessionWpm then
		sessionWpm = math.random(lo, hi)
		currentWpm = sessionWpm
		return
	end

	currentWpm = math.clamp(currentWpm + (sessionWpm - currentWpm) * 0.3 + math.random(-4, 4), lo, hi)
end

local function curveShape(t, shape)
	t = math.clamp(t, 0, 1)

	if shape == "Exp" then
		return (math.exp(2 * t) - 1) / (math.exp(2) - 1)
	elseif shape == "Log" then
		return math.log(1 + 2 * t) / math.log(3)
	end

	return t
end

local function thinkTime(prefix)
	local lo = math.min(Options.StartMin.Value, Options.StartMax.Value)
	local hi = math.max(Options.StartMin.Value, Options.StartMax.Value)
	local base = lo + math.random() * (hi - lo)

	if not Toggles.ThinkCurve.Value then
		return base
	end

	local driver = Options.ThinkDriver.Value
	local level

	if driver == "Words used" then
		level = usedCount
	elseif driver == "Turn number" then
		level = turnCount
	else
		level = 60 - math.min(#findWords(prefix, false, false, true), 60)
	end

	local extra = Options.ThinkExtra.Value * curveShape((level - Options.ThinkThreshold.Value) / 20, Options.ThinkShape.Value)

	return math.clamp(base + extra, base, Options.ThinkCap.Value)
end

local humanDelays = IB_NO_VIRTUALIZE(function(word, from)
	local delays = {}
	local total = 0
	local prev = word:sub(from - 1, from - 1)

	for i = from, #word do
		local c = word:sub(i, i)
		local d = 1.2
		local a = keys[c]
		local b = keys[prev]

		if a and b then
			if c == prev then
				d = 0.2
			elseif a.h ~= b.h then
				d = 0.5
			else
				d = 0.8 + math.sqrt((a.x - b.x) ^ 2 + (a.y - b.y) ^ 2) * 0.3
			end
		end

		d *= math.random(90, 110) / 100
		delays[#delays + 1] = d
		total += d
		prev = c
	end

	local target = #delays * perCharDelay()

	if #delays < 6 then
		target = math.clamp(target, 0.1, 0.7)
	else
		target = math.clamp(target, 0.28, 2.5)
	end

	for i, v in delays do
		delays[i] = v / total * target
	end

	return delays
end)

local submitNext = IB_NO_VIRTUALIZE(function(token)
	if unloaded or token ~= roundId or not Toggles.AutoAnswer.Value or turnPlayer ~= player then
		return
	end

	task.wait()

	if token ~= roundId then
		return
	end

	if not (currentPrompt and currentPrompt.RequiredLetter) then
		return
	end

	local prefix = string.lower(currentPrompt.RequiredLetter)
	local word = pickCandidate(prefix)

	if not word then
		Library:Notify(`no words left for "{prefix}"`, 3)
		return
	end

	triedThisRound[word] = true
	pendingWord = nil

	local delays

	if Toggles.HumanType.Value and not Toggles.InstantAnswer.Value then
		delays = humanDelays(word, #prefix + 1)
	end

	if retries == 0 and not Toggles.InstantAnswer.Value then
		local preDelay = thinkTime(prefix)

		if Toggles.HumanType.Value then
			preDelay *= math.random(85, 120) / 100
		end

		local waited = 0

		while token == roundId and waited < preDelay do
			task.wait(0.05)
			waited += 0.05
		end

		if token ~= roundId then
			return
		end
	end

	local prompt = findPrompt()
	local input = gameInput

	if not prompt or prompt.Choices or prompt.RequiredLetter ~= currentPrompt.RequiredLetter or not input then
		return
	end

	local typed = string.lower(input:currentWord())

	if typed:sub(1, #prefix) ~= prefix then
		return
	end

	for _ = #prefix + 1, #typed do
		if token ~= roundId then
			return
		end

		event.fire("tryKeystroke", -1)

		if Toggles.HumanType.Value and not Toggles.InstantAnswer.Value then
			task.wait(math.random(9, 14) / 100)
		end
	end

	local typos = Toggles.Typos.Value and not Toggles.InstantAnswer.Value and successCount >= Options.TypoWarmup.Value
	local typoChance = 0

	if typos and #word > Options.TypoSafeLen.Value then
		typoChance = Options.TypoChance.Value * (1 + (#word - Options.TypoSafeLen.Value) * 0.4)
	end

	for i = #prefix + 1, #word do
		if token ~= roundId then
			return
		end

		local ch = word:sub(i, i)

		if typoChance > 0 and math.random(1, 1000) <= typoChance * 10 then
			local wrong = adjacentKey(ch)

			if wrong then
				event.fire("tryKeystroke", string.upper(wrong))
				task.wait(math.random(180, 320) / 1000)

				if token ~= roundId then
					return
				end

				event.fire("tryKeystroke", -1)
				task.wait(math.random(60, 120) / 1000)

				if token ~= roundId then
					return
				end
			end
		end

		event.fire("tryKeystroke", string.upper(ch))

		if not Toggles.InstantAnswer.Value then
			task.wait(delays and delays[i - #prefix] or perCharDelay())
		end
	end

	if token ~= roundId then
		return
	end

	if not Toggles.InstantAnswer.Value then
		task.wait(Options.AnswerDelay.Value)
	end

	if token ~= roundId or string.lower(input:currentWord()) ~= word then
		return
	end

	submitAt = os.clock()
	pendingWord = word
	event.fire("tryAnswer")
end)

local bestChoice = IB_NO_VIRTUALIZE(function(choices)
	local best
	local bestN = -1

	for _, v in choices do
		local n = #findWords(string.lower(tostring(v)), false, false, true)

		if n > bestN then
			best = v
			bestN = n
		end
	end

	return best
end)

local function applyShakeHide(v)
	if v then
		if shakeConn or not uis.AccelerometerEnabled then
			return
		end

		local lastShake = 0
		local spikes = 0
		local windowStart = 0

			if os.clock() - lastShake < 1 then
				return
			end

			if accel.Position.Magnitude < 3.5 then
				return
			end

			local now = os.clock()

			if now - windowStart > 0.6 then
				spikes = 0
				windowStart = now
			end

			spikes += 1

			if spikes >= 2 then
				spikes = 0
				lastShake = now
				Library:Toggle()
			end
		end)
	elseif shakeConn then
		shakeConn:Disconnect()
		shakeConn = nil
	end

refreshSuggestions = IB_NO_VIRTUALIZE(function(prompt)
	if not wordsReady then
		setStatus("words still downloading...")
		return
	end

	if prompt.Choices then
		local rows = {}

		for _, v in prompt.Choices do
			rows[#rows + 1] = {text = `{string.upper(tostring(v))}: {#findWords(string.lower(tostring(v)), false, false, true)} words`}
		end

		setRows(rows)
	elseif prompt.RequiredLetter then
		local prefix = string.lower(prompt.RequiredLetter)
		local items = orderedCandidates(prefix, not Toggles.RemoveUsed.Value, false)
		local rows = {}

		for _, v in items do
			if #rows >= Options.SuggestCount.Value then
				break
			end

			rows[#rows + 1] = {text = v.word, check = v.check}
		end

		if #rows > 0 then
			setRows(rows)
		elseif #suffixList() > 0 then
			setStatus(`no words with those endings for "{prefix}"`)
		else
			setStatus(`no words for "{prefix}"`)
		end
	end
end)

local function onRound(prompt, turnPlr)
	roundId += 1
	currentPrompt = prompt
	turnPlayer = turnPlr
	triedThisRound = {}
	retries = 0
	pendingWord = nil

	refreshSuggestions(prompt)

	if turnPlr ~= player or not wordsReady then
		return
	end

	turnCount += 1
	driftWpm()

	if prompt.Choices then
		if Toggles.AutoChoose.Value then
			local pick = bestChoice(prompt.Choices)

			if pick then
				local token = roundId

				task.spawn(function()
					if not unloaded and token == roundId and Toggles.AutoChoose.Value then
						event.remoteFire("chooseLetter", string.lower(tostring(pick)))
					end
				end)
			end
		end

		return
	end

	if prompt.RequiredLetter and Toggles.AutoAnswer.Value then
		local token = roundId
		task.spawn(function() submitNext(token) end)
	end
end

resumeRound = function()
	if unloaded or not player:GetAttribute("IsTurn") then
		return
	end

	local prompt = findPrompt()

	if prompt then
		onRound(prompt, player)
	end
end

Toggles.AutoAnswer:OnChanged(function()
	if Toggles.AutoAnswer.Value then
		if wordsReady then

	else
		roundId += 1
		pendingWord = nil
	end
end)

local function onCorrect(word)
	local w

	if type(word) == "string" and string.lower(word):match("^[a-z]+$") then
		w = string.lower(word)

		if not usedWords[w] then
			usedWords[w] = true
			usedCount += 1
		end
	end

	if pendingWord and w == pendingWord then
		successCount += 1
	end

	if Toggles.AutoLearn.Value and w and (not whitelist[w] or blacklist[w]) then
		addWhitelist(w)
	end

	roundId += 1
	pendingWord = nil
end

local function petCutsceneActive()
	return os.clock() - lastAttackAt < 1.5 or workspace.CurrentCamera.CameraType == Enum.CameraType.Scriptable
end

local function onStrike(uid, alreadyUsed)
	if uid ~= player.UserId then
		return
	end

	if not (pendingWord and Toggles.AutoAnswer.Value) then
		return
	end

	if petCutsceneActive() then
		return
	end

	local w = pendingWord
	pendingWord = nil
	roundId += 1

	if alreadyUsed then
		if not usedWords[w] then
			usedWords[w] = true
			usedCount += 1
		end
	elseif Toggles.AutoLearn.Value and not blacklist[w] and os.clock() - submitAt < 3 then
		addBlacklist(w)

		if currentPrompt then
			refreshSuggestions(currentPrompt)
		end
	end

	retries += 1

	if retries >= Options.MaxRetries.Value then
		return
	end

	local token = roundId

	task.spawn(function()
		submitNext(token)
	end)
end

local bridge = Instance.new("BindableEvent")

bridge.Event:Connect(function(kind, ...)
	if kind == "round" then
		onRound(...)
	elseif kind == "strike" then
		onStrike(...)
	elseif kind == "correct" then
		onCorrect(...)
	end
end)

local roundConn = event.remoteConnect("updateRound", function(prompt, _, turnPlr)
	if type(prompt) ~= "table" then
		return
	end

	bridge:Fire("round", prompt, turnPlr)
end)

local correctConn = event.remoteConnect("correct", function(word)
	bridge:Fire("correct", word)
end)

local strikeConn = event.remoteConnect("strike", function(uid, _, alreadyUsed)
	bridge:Fire("strike", uid, alreadyUsed)
end)

local damageConn = event.remoteConnect("takeDamage", function(uid)
	if uid ~= player.UserId then
		return
	end

	roundId += 1
	pendingWord = nil
end)

local attackConn = event.remoteConnect("attack", function()
	lastAttackAt = os.clock()
end)

task.defer(resumeRound)

inGameConn = player:GetAttributeChangedSignal("InGame"):Connect(function()
	if player:GetAttribute("InGame") then
		usedWords = {}
		usedCount = 0
		turnCount = 0
	end
end)

Library:OnUnload(function()
	unloaded = true
	roundId += 1

	event.disconnect(roundConn)
	event.disconnect(correctConn)
	event.disconnect(strikeConn)
	event.disconnect(damageConn)
	event.disconnect(attackConn)

	bridge:Destroy()

	if afkConn then
		afkConn:Disconnect()
		afkConn = nil
	end

	if inGameConn then
		inGameConn:Disconnect()
		inGameConn = nil
	end

	if shakeConn then
		shakeConn:Disconnect()
		shakeConn = nil
	end

	if gameCfg and afkThreshold then
		nudgeInput()
		gameCfg.AFK.Threshold = afkThreshold
	end

	getgenv().FinishTheWordUnload = nil
end)
