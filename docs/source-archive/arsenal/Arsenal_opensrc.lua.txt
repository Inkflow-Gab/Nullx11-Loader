local function new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

local function corner(parent, r)
    return new("UICorner", {CornerRadius = UDim.new(0, r or 8)}, parent)
end

    return new("UIStroke", {
        Color = color or C.accent,
        Thickness = thick or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,

end

local function gradient(parent, c1, c2, rot)
    return new("UIGradient", {
        Color = ColorSequence.new(c1, c2),
        Rotation = rot or 90,
    }, parent)
end

local function setMinimized(state)
    minimized = state

        sidebar.Visible = false

local dragging, dragStart, startPos = false, nil, nil
local lastTouchTime = 0

top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = win.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then 
                dragging = false 
                if input.UserInputType == Enum.UserInputType.Touch then
                    local dt = tick() - lastTouchTime
                    if dt < 0.2 then setMinimized(not minimized) end
                    lastTouchTime = tick()
                end
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        win.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + d.X,
            startPos.Y.Scale, startPos.Y.Offset + d.Y
        )
    end
end)

UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == Enum.KeyCode.K then
        setMinimized(not minimized)
    end
end)

local function makeTab(name)
    local btn = new("TextButton", {
        Size = UDim2.new(1, 0, 0, 34),
        BackgroundColor3 = C.bg,
        Text = "  " .. name,
        Font = Enum.Font.GothamSemibold,

    }, page)
    new("UIPadding", {
        PaddingTop = UDim.new(0, 4),
        PaddingBottom = UDim.new(0, 16),
        PaddingRight = UDim.new(0, 12),
    }, page)

    btn.MouseEnter:Connect(function()
        if activeTab ~= name then
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = C.panel, TextColor3 = C.text}):Play()
        end
    end)
    btn.MouseLeave:Connect(function()
        if activeTab ~= name then
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = C.bg, TextColor3 = C.muted}):Play()
        end
    end)

    tabs[name] = {btn = btn, bar = bar, page = page}

    btn.MouseButton1Click:Connect(function()
        if activeTab == name then return end
        for n, t in pairs(tabs) do
            if n == name then
                t.page.Visible = true
                TweenService:Create(t.btn, TweenInfo.new(0.2), {BackgroundColor3 = C.accent, TextColor3 = C.text}):Play()
                TweenService:Create(t.bar, TweenInfo.new(0.2), {Size = UDim2.new(0, 3, 0.6, 0)}):Play()

                t.page.CanvasPosition = Vector2.zero
                TweenService:Create(t.btn, TweenInfo.new(0.15), {BackgroundColor3 = C.bg, TextColor3 = C.muted}):Play()
                TweenService:Create(t.bar, TweenInfo.new(0.15), {Size = UDim2.new(0, 3, 0, 0)}):Play()
            end
        end
        activeTab = name
    end)

    return page
end

local espSettings = {
    enabled  = false,
    hitbox   = true,
    skeleton = true,
    name     = true,
    distance = true,
    color    = C.accent,
    maxDist  = 2000,
}

local ESP = {drawings = {}, connections = {}}

local function getDrawing() return Drawing or (getgenv and getgenv().Drawing) or nil end

local function newDrawing(class, props)
    local d = getDrawing()
    if not d then return nil end
    local obj = d.new(class)
    for k, v in pairs(props or {}) do obj[k] = v end
    obj.Visible = false
    return obj
end

local function clearESP()
    for _, set in pairs(ESP.drawings) do
        for k, v in pairs(set) do
            if k == "bones" then
                for _, b in pairs(v) do pcall(function() b:Remove() end) end
            else
                pcall(function() v:Remove() end)
            end
        end
    end
    ESP.drawings = {}
end

local SKELETON = {
    {"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"}, {"LowerTorso", "HumanoidRootPart"},
    {"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"},
    {"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
    {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
    {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"},
    {"Head", "Torso"}, {"Torso", "HumanoidRootPart"},
    {"Torso", "Left Arm"}, {"Left Arm", "Left Arm"},
    {"Torso", "Right Arm"},
    {"Torso", "Left Leg"}, {"Left Leg", "Left Leg"},
    {"Torso", "Right Leg"},
}

local function createESPFor(char)
    if not getDrawing() then return end
    if not char:FindFirstChild("HumanoidRootPart") then return end
    local set = {
        hitbox = newDrawing("Square", {Thickness = 1.5, Filled = false}),
        name   = newDrawing("Text", {Size = 13, Center = true, Outline = true, Font = 2}),
        dist   = newDrawing("Text", {Size = 12, Center = true, Outline = true, Font = 2}),
        bones  = {},
    }
    for i = 1, #SKELETON do
        set.bones[i] = newDrawing("Line", {Thickness = 1.2})
    end
    ESP.drawings[char] = set
end

local function destroyESPFor(char)
    local set = ESP.drawings[char]
    if not set then return end
    for k, v in pairs(set) do
        if k == "bones" then
            for _, b in pairs(v) do pcall(function() b:Remove() end) end
        else
            pcall(function() v:Remove() end)
        end
    end
    ESP.drawings[char] = nil
end

-- sweep dead character references every second
task.spawn(function()
    while gui.Parent do
        for char, _ in pairs(ESP.drawings) do
            if not char or not char.Parent or not char:FindFirstChild("HumanoidRootPart") or not char:FindFirstChildOfClass("Humanoid") or char:FindFirstChildOfClass("Humanoid").Health <= 0 then
                destroyESPFor(char)
            end
        end
        task.wait(1)
    end
end)

local function updateESP()
    local cam = workspace.CurrentCamera
    if not cam then return end
    local origin = cam.CFrame.Position

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local skip = false
            if plr.Team and LocalPlayer.Team and plr.Team == LocalPlayer.Team then skip = true end
            
            local char = plr.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                if skip or not espSettings.enabled then
                    if ESP.drawings[char] then
                        for k, v in pairs(ESP.drawings[char]) do
                            if k == "bones" then for _, b in pairs(v) do b.Visible = false end
                            else v.Visible = false end
                        end
                    end
                else
                    if not ESP.drawings[char] then createESPFor(char) end
                    local set = ESP.drawings[char]
                    if set then
                        local root = char.HumanoidRootPart
                        local dist = (root.Position - origin).Magnitude
                        if dist > espSettings.maxDist then
                            for k, v in pairs(set) do
                                if k == "bones" then for _, b in pairs(v) do b.Visible = false end
                                else v.Visible = false end
                            end
                        else
                            local top, on = cam:WorldToViewportPoint(root.Position + Vector3.new(0, 2.5, 0))
                            local bot = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 2.5, 0))
                            local h = math.abs(top.Y - bot.Y)
                            local w = h * 0.55
                            local bx = top.X - w / 2
                            local by = top.Y

                            if set.hitbox and espSettings.hitbox then
                                set.hitbox.Visible = on
                                set.hitbox.Position = Vector2.new(bx, by)
                                set.hitbox.Size = Vector2.new(w, h)
                                set.hitbox.Color = espSettings.color
                            elseif set.hitbox then set.hitbox.Visible = false end

                            if set.name and espSettings.name then
                                set.name.Visible = on
                                set.name.Position = Vector2.new(top.X, by - 14)
                                set.name.Text = plr.Name
                                set.name.Color = C.text
                            elseif set.name then set.name.Visible = false end

                            if set.dist and espSettings.distance then
                                set.dist.Visible = on
                                set.dist.Position = Vector2.new(top.X, bot.Y + 4)
                                set.dist.Text = string.format("[%d]", math.floor(dist))
                                set.dist.Color = C.muted
                            elseif set.dist then set.dist.Visible = false end

                            if set.bones and espSettings.skeleton then
                                for i, pair in ipairs(SKELETON) do
                                    local a = char:FindFirstChild(pair[1])
                                    local b = char:FindFirstChild(pair[2])
                                    local line = set.bones[i]
                                    if line then
                                        if a and b and a:IsA("BasePart") and b:IsA("BasePart") then
                                            local p1, o1 = cam:WorldToViewportPoint(a.Position)
                                            local p2, o2 = cam:WorldToViewportPoint(b.Position)
                                            if o1 and o2 then
                                                line.Visible = on
                                                line.From = Vector2.new(p1.X, p1.Y)
                                                line.To = Vector2.new(p2.X, p2.Y)
                                                line.Color = espSettings.color
                                            else
                                                line.Visible = false
                                            end
                                        else
                                            line.Visible = false
                                        end
                                    end
                                end
                            else
                                if set.bones then for _, l in pairs(set.bones) do l.Visible = false end end
                            end
                        end
                    end
                end
            end
        end
    end
end

local function espWatcher()
    ESP.connections = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            table.insert(ESP.connections, plr.CharacterAdded:Connect(function(c)
                task.wait(0.5)
                createESPFor(c)
            end))
            if plr.Character then createESPFor(plr.Character) end
            table.insert(ESP.connections, plr.CharacterRemoving:Connect(function(c)
                destroyESPFor(c)
            end))
        end
    end
    table.insert(ESP.connections, Players.PlayerRemoving:Connect(function(plr)
        if plr.Character then destroyESPFor(plr.Character) end
    end))
end

--== Aimbot ==--
local aimSettings = {
    enabled    = false,
    fov        = 150,
    smooth     = 0.25,
    maxDist    = 500,
    visible    = true,
    showFov    = false,
}

local fovCircle = nil

local function updateFovCircle()
    if not aimSettings.showFov or not aimSettings.enabled then
        if fovCircle then fovCircle.Visible = false end
        return
    end

    if not fovCircle then
        fovCircle = newDrawing("Circle", {Thickness = 1, Filled = false})
    end

    if fovCircle then
        local cam = workspace.CurrentCamera
        if cam then
            local vp = cam.ViewportSize
            fovCircle.Visible = true
            fovCircle.Position = Vector2.new(vp.X / 2, vp.Y / 2)
            fovCircle.Radius = aimSettings.fov
            fovCircle.Color = Color3.fromRGB(255, 255, 255)
        end
    end
end

local function getClosestTarget()
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local vp = cam.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    local closest, dist = nil, aimSettings.fov

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local isTeammate = plr.Team and LocalPlayer.Team and plr.Team == LocalPlayer.Team
            if not isTeammate then
                local char = plr.Character
                if char then
                    local head = char:FindFirstChild("Head")
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    
                    if head and hum and hum.Health > 0 then
                        local worldDist = (head.Position - cam.CFrame.Position).Magnitude
                        if worldDist <= aimSettings.maxDist then
                            local screen, onScreen = cam:WorldToViewportPoint(head.Position)
                            if onScreen then
                                local screenDist = (Vector2.new(screen.X, screen.Y) - center).Magnitude
                                if screenDist < dist then
                                    local canSee = true
                                    if aimSettings.visible then
                                        local rayParams = RaycastParams.new()
                                        rayParams.FilterType = Enum.RaycastFilterType.Exclude
                                        rayParams.FilterDescendantsInstances = {LocalPlayer.Character, cam}
                                        local result = workspace:Raycast(cam.CFrame.Position, head.Position - cam.CFrame.Position, rayParams)
                                        if result and result.Instance then
                                            if not result.Instance:IsDescendantOf(char) then
                                                canSee = false
                                            end
                                        end
                                    end

                                    if canSee then
                                        closest = head
                                        dist = screenDist
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return closest
end

local function aimLoop()
    updateFovCircle()
    if not aimSettings.enabled then return end
    local target = getClosestTarget()
    if target then
        local cam = workspace.CurrentCamera
        local currentCF = cam.CFrame
        local targetCF = CFrame.new(currentCF.Position, target.Position)
        cam.CFrame = currentCF:Lerp(targetCF, aimSettings.smooth)
    end
end

--== Movement ==--
local moveSettings = {
    speedEnabled = false,
    speed = 16,
    jumpEnabled = false,
    jump = 50,
    fly = false,
    flySpeed = 50
}

local flyConnection = nil
local bodyVelocity = nil
local bodyGyro = nil
local flying = false

RunService.RenderStepped:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and not moveSettings.fly then
        if moveSettings.speedEnabled then
            hum.WalkSpeed = moveSettings.speed
        else
            hum.WalkSpeed = 16
        end
        if moveSettings.jumpEnabled then
            hum.UseJumpPower = true
            hum.JumpPower = moveSettings.jump
        end
    end
end)

local function stopFly()
    flying = false
    if flyConnection then flyConnection:Disconnect(); flyConnection = nil end
    local char = LocalPlayer.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if root then
            if bodyVelocity then bodyVelocity:Destroy(); bodyVelocity = nil end
            if bodyGyro then bodyGyro:Destroy(); bodyGyro = nil end
        end
        if hum then
            hum.PlatformStand = false
        end
    end
end

local function startFly()
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    flying = true
    hum.PlatformStand = true

    bodyVelocity = new("BodyVelocity", {
        P = Vector3.new(1e4, 1e4, 1e4),
        MaxForce = Vector3.new(1e4, 1e4, 1e4),
        Velocity = Vector3.new(0, 0, 0)
    }, root)

    bodyGyro = new("BodyGyro", {
        P = 1e4,
        MaxTorque = Vector3.new(1e4, 1e4, 1e4),
        CFrame = root.CFrame
    }, root)

    flyConnection = RunService.RenderStepped:Connect(function()
        if not flying or not root or not bodyVelocity or not bodyGyro then return end
        local cam = workspace.CurrentCamera
        local moveDir = Vector3.zero

        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir += cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir -= cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir -= cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir += cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir += Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir -= Vector3.new(0, 1, 0) end

        bodyVelocity.Velocity = moveDir * moveSettings.flySpeed
        bodyGyro.CFrame = cam.CFrame
    end)
end

local function setFly(enabled)
    moveSettings.fly = enabled
    if enabled then
        startFly()
    else
        stopFly()
    end
end
